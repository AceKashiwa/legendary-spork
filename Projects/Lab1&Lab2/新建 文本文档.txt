module mux2(
    input A,
    input B,
    input S,
    output Y 
    );
endmodule

module mux2to1(A, B, S, Y);
    input A, B, S;
    output Y;
    // Boolean function Verilog Description
    assign Y = S & B + (~S) & A;
endmodule

module mux2to1_g(A, B, S, Y);
    input A, B, S;
    output Y;
    // AND/OR/NOT gate level Verilog Description
    wire Y1, Y2, notS;
    and a1(Y1, A, notS);
    and a2(Y2, B, S);
    not (notS, S);
    or(Y, A, B);
endmodule

module mux2to1_a(A, B, S, Y);
    input A, B, S;
    output Y;
    // Always statement and if-else statement
    reg Y;
        always @(*)
            begin
                if (S==0) Y = A;
                else Y = B; 
            end
endmodule

module mux2to1_c(A, B, S, Y);
    input A, B, S;
    output Y;
    // always statement and case statement
    reg Y;
        always @(*)
            begin
                case (S)
                1'b0: Y=A;
                1'b1: Y=B;
                endcase
            end
endmodule

module mux4to1(D0, D1, D2, D3, S, Y);
    input D0, D1, D2, D3;
    input [2:0] S;
    output Y;

    reg Y;  
    reg mux1out, mux2out;
    always @(*)
        begin
            if (S[0]==0)
                mux2to1_g mux1(D0, D2, S[1], mux1out);
            else
                mux2to1_g mux2(D1, D3, S[1], mux2out);
        end
endmodule
